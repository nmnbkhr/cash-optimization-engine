### Step 2: Create a thin wrapper — backend/app/core/sbp_data.py

This wraps sbp_easydata.py functions into what business_output.py needs.
It does NOT duplicate the scraper — it CALLS it.

```python
"""
SBP Live Data Service for COE
Wraps the pakfindata sbp_easydata.py scraper.
Provides convenience methods for business_output.py.
Falls back to constants if API/cache unavailable.
"""
from app.core.sbp_easydata import (
    get_kibor, get_policy_rate, get_daily_fx, 
    get_exchange_rate, get_walr, get_cpi,
    read_series, sync_all_to_db
)
from app.core.constants import SBP_POLICY_RATE
from datetime import date, timedelta
import logging

logger = logging.getLogger(__name__)


class SBPDataService:
    """
    Thin layer over pakfindata scraper.
    3 modes:
    1. LIVE: calls EasyData API directly (rate-limited, 15s delay)
    2. CACHED: reads from pre-downloaded CSVs in data/sbp_cache/series/
    3. FALLBACK: returns constants.py values
    
    For COE app startup: use CACHED mode (no API delay).
    For daily refresh: run `python -m app.core.sbp_easydata update` via cron.
    For first-time setup: run `python -m app.core.sbp_easydata fetch-priority`.
    """
    
    def get_overnight_kibor(self) -> float:
        """Latest overnight KIBOR offer rate. THE key rate for opportunity cost."""
        try:
            # Try cached CSV first (no API call)
            data = read_series("TS_GP_BAM_SIRKIBOR_D.5KIBOR6M")  # 6M KIBOR as proxy
            if data:
                latest = data[-1]["value"]
                return latest / 100  # stored as percentage
            
            # Try live API (15s delay)
            rates = get_kibor(tenor="6M", start_date=(date.today() - timedelta(days=30)).isoformat())
            if rates:
                return rates[-1][1] / 100
        except Exception as e:
            logger.warning(f"KIBOR fetch failed: {e}")
        
        return SBP_POLICY_RATE  # fallback
    
    def get_policy_rate_current(self) -> float:
        """Current SBP policy rate."""
        try:
            data = read_series("TS_GP_IR_SIRPR_AH.SBPOL0030")
            if data:
                return data[-1]["value"] / 100
            
            rates = get_policy_rate(start_date=(date.today() - timedelta(days=90)).isoformat())
            if rates:
                return rates[-1][1] / 100
        except Exception as e:
            logger.warning(f"Policy rate fetch failed: {e}")
        
        return SBP_POLICY_RATE
    
    def get_fx_rate(self, currency: str = "USD") -> float:
        """Latest PKR per unit of foreign currency."""
        try:
            data = get_daily_fx(currency=currency, start_date=(date.today() - timedelta(days=30)).isoformat())
            if data:
                return data[-1][1]
        except Exception as e:
            logger.warning(f"FX rate fetch failed for {currency}: {e}")
        
        # Fallback rates
        fallback = {"USD": 278.50, "EUR": 305.0, "GBP": 355.0, "AED": 75.85,
                    "SAR": 74.20, "JPY": 1.85, "CNY": 38.50, "CHF": 320.0}
        return fallback.get(currency.upper(), 278.50)
    
    def get_kibor_curve(self) -> dict:
        """Full KIBOR yield curve — all tenors."""
        tenors = {"1W": "1W", "2W": "2W", "1M": "1M", "3M": "3M", 
                  "6M": "6M", "9M": "9M", "12M": "1Y"}
        curve = {}
        for label, tenor in tenors.items():
            try:
                rates = get_kibor(tenor=tenor, start_date=(date.today() - timedelta(days=30)).isoformat())
                if rates:
                    curve[label] = rates[-1][1]
            except:
                pass
        return curve
    
    def get_weighted_avg_lending(self) -> float:
        """Industry weighted average lending rate."""
        try:
            rates = get_walr(start_date=(date.today() - timedelta(days=90)).isoformat())
            if rates:
                return rates[-1][1] / 100
        except:
            pass
        return 0.15  # fallback ~15%
    
    def get_cpi_latest(self) -> float:
        """Latest CPI inflation YoY %."""
        try:
            data = get_cpi(start_date=(date.today() - timedelta(days=90)).isoformat())
            if data:
                return data[-1][1]
        except:
            pass
        return 10.0  # fallback
    
    def get_all_rates_summary(self) -> dict:
        """Everything in one call — for dashboard display."""
        return {
            "as_of": str(date.today()),
            "kibor_overnight": self.get_overnight_kibor(),
            "policy_rate": self.get_policy_rate_current(),
            "kibor_curve": self.get_kibor_curve(),
            "fx": {
                "USD": self.get_fx_rate("USD"),
                "EUR": self.get_fx_rate("EUR"),
                "GBP": self.get_fx_rate("GBP"),
                "AED": self.get_fx_rate("AED"),
                "SAR": self.get_fx_rate("SAR"),
            },
            "weighted_avg_lending": self.get_weighted_avg_lending(),
            "cpi_yoy": self.get_cpi_latest(),
        }


# Singleton
_service = None
def get_sbp_service() -> SBPDataService:
    global _service
    if _service is None:
        _service = SBPDataService()
    return _service
```

### HOW IT WORKS IN THE APP

```
sbp_easydata.py (from pakfindata)
  ├── API calls → easydata.sbp.org.pk/api/v1
  ├── Saves → data/sbp_cache/series/*.csv + *.json
  └── Rate limited (15s between calls, 240/hr)
        │
        ▼
sbp_data.py (thin wrapper)
  ├── read_series() → reads cached CSVs (NO API call, instant)
  ├── get_kibor() → API only if cache miss
  └── fallback → constants.py if everything fails
        │
        ▼
business_output.py
  ├── self.kibor_overnight = sbp.get_overnight_kibor()
  ├── self.policy_rate = sbp.get_policy_rate_current()
  └── self.fx_rates = {...}
        │
        ▼
All UC calculations use LIVE KIBOR instead of hardcoded 0.105
```
